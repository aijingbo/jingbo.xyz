## 3. Three.js 场景搭建规范

```javascript
// 基本三件套
const scene = new THREE.Scene();
scene.background = new THREE.Color(0x0f172a); // 深色背景
scene.fog = new THREE.Fog(0x0f172a, 8, 25);

const camera = new THREE.PerspectiveCamera(50, aspect, 0.1, 100);
camera.position.set(5, 5, 5);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setPixelRatio(window.devicePixelRatio);
renderer.setSize(width, height);
container.appendChild(renderer.domElement);

// 控制器
const controls = new OrbitControls(camera, renderer.domElement);
controls.enableDamping = true;
controls.dampingFactor = 0.08;

// 光照
const ambient = new THREE.AmbientLight(0xffffff, 0.5);
const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
dirLight.position.set(5, 10, 5);
scene.add(ambient, dirLight);

// 渲染循环
function animate() {
  requestAnimationFrame(animate);
  controls.update();
  renderer.render(scene, camera);
}
animate();

// 响应式
window.addEventListener('resize', () => {
  camera.aspect = container.clientWidth / container.clientHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(container.clientWidth, container.clientHeight);
});
```

### 3D 元素透明度过渡（步骤高亮用）

```javascript
// 平滑过渡元素透明度
function tweenOpacity(mesh, target, duration = 400) {
  const start = mesh.material.opacity;
  const startTime = performance.now();
  function step() {
    const t = Math.min((performance.now() - startTime) / duration, 1);
    mesh.material.opacity = start + (target - start) * t;
    mesh.material.transparent = true;
    if (t < 1) requestAnimationFrame(step);
  }
  step();
}
```

### 3D 文字标签（顶点标注）

```javascript
// 用 CSS2DRenderer 或 Sprite + Canvas 纹理
function makeLabel(text, color = '#ffffff') {
  const canvas = document.createElement('canvas');
  canvas.width = 128; canvas.height = 64;
  const ctx = canvas.getContext('2d');
  ctx.fillStyle = color;
  ctx.font = 'bold 36px sans-serif';
  ctx.textAlign = 'center';
  ctx.fillText(text, 64, 42);
  const texture = new THREE.CanvasTexture(canvas);
  const material = new THREE.SpriteMaterial({ map: texture, transparent: true });
  const sprite = new THREE.Sprite(material);
  sprite.scale.set(0.8, 0.4, 1);
  return sprite;
}
```

## 4. Canvas 2D 绘制规范

```javascript
const canvas = document.getElementById('board-container');
const ctx = canvas.getContext('2d');

// 高 DPI 适配
function resizeCanvas() {
  const dpr = window.devicePixelRatio || 1;
  const rect = canvas.parentElement.getBoundingClientRect();
  canvas.width = rect.width * dpr;
  canvas.height = rect.height * dpr;
  canvas.style.width = rect.width + 'px';
  canvas.style.height = rect.height + 'px';
  ctx.scale(dpr, dpr);
}

// 坐标系转换（数学坐标 → 屏幕坐标）
const cx = canvas.clientWidth / 2;
const cy = canvas.clientHeight / 2;
const scale = 50; // 1 单位 = 50 像素

function toScreen(x, y) {
  return [cx + x * scale, cy - y * scale];
}

// 绘制网格
function drawGrid() {
  ctx.strokeStyle = '#e2e8f0';
  ctx.lineWidth = 1;
  // ... 网格线
}

// 绘制函数曲线
function drawCurve(fn, color = '#0ea5e9') {
  ctx.strokeStyle = color;
  ctx.lineWidth = 2;
  ctx.beginPath();
  for (let px = 0; px <= canvas.clientWidth; px++) {
    const x = (px - cx) / scale;
    const y = fn(x);
    const [sx, sy] = toScreen(x, y);
    if (px === 0) ctx.moveTo(sx, sy);
    else ctx.lineTo(sx, sy);
  }
  ctx.stroke();
}
```

## 5. 代码规范

- **HTML**：每个 Demo 是单个自包含文件，内联 CSS 和 JS
- **JS**：使用 ES Module（Three.js）或普通 script（Canvas 2D）
- **CSS**：使用 CSS 变量系统，不使用预处理器
- **命名**：文件名 kebab-case，CSS 类名 BEM-like
- **注释**：所有代码注释用中文
- **公式**：行内用 `$...$`，行间用 `$$...$$`
- **性能**：3D 场景保持 60fps，滑块响应 < 16ms
- **体积**：单个 Demo HTML < 200KB（不含 CDN 库）

## 6. 3D 场景背景色

- 3D Demo 使用深色背景：`scene.background = new THREE.Color(0x0f172a)`
- 2D Demo 使用浅色背景：白色或 `#f8fafc`
- 化学 Demo 可用更深的背景：`#0a0a1a`

## 7. 重要提醒

- Three.js 必须通过 **import maps** 加载，不能直接用 `<script>` 标签
- 使用 import maps 的 Demo 必须通过 HTTP 服务器访问（不能 file:// 双击打开）
- KaTeX 的 auto-render 需要在 DOM 加载后调用
- MathJax 配置必须在加载 MathJax 脚本之前设置
